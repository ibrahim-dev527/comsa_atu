🗄️ Step 2 — Run the Complete SQL
Go to Supabase → SQL Editor → New Query. Paste and run this entire block:
-- COMSA: Drop and recreate all tables DROP TABLE IF EXISTS gallery_photos CASCADE; DROP TABLE IF EXISTS past_questions CASCADE; DROP TABLE IF EXISTS posts CASCADE; DROP TABLE IF EXISTS events CASCADE; CREATE TABLE events ( id uuid DEFAULT gen_random_uuid() PRIMARY KEY, title text NOT NULL, description text, category text, event_date date NOT NULL, event_time text, venue text, audience text, status text DEFAULT 'upcoming', icon text DEFAULT '🎉', image_url text, week_group text, created_at timestamptz DEFAULT now() ); CREATE TABLE posts ( id uuid DEFAULT gen_random_uuid() PRIMARY KEY, title text NOT NULL, excerpt text, content text, category text DEFAULT 'news', author text DEFAULT 'COMSA Secretariat', image_url text, is_featured boolean DEFAULT false, tags text[], created_at timestamptz DEFAULT now() ); CREATE TABLE past_questions ( id uuid DEFAULT gen_random_uuid() PRIMARY KEY, course_title text NOT NULL, course_code text, year_sem text, programme text, pdf_url text, storage_path text, created_at timestamptz DEFAULT now() ); CREATE TABLE gallery_photos ( id uuid DEFAULT gen_random_uuid() PRIMARY KEY, caption text, image_url text, storage_path text, created_at timestamptz DEFAULT now() ); ALTER TABLE events ENABLE ROW LEVEL SECURITY; ALTER TABLE posts ENABLE ROW LEVEL SECURITY; ALTER TABLE past_questions ENABLE ROW LEVEL SECURITY; ALTER TABLE gallery_photos ENABLE ROW LEVEL SECURITY; CREATE POLICY "Public read events" ON events FOR SELECT USING (true); CREATE POLICY "Public read posts" ON posts FOR SELECT USING (true); CREATE POLICY "Public read past_questions" ON past_questions FOR SELECT USING (true); CREATE POLICY "Public read gallery_photos" ON gallery_photos FOR SELECT USING (true); CREATE POLICY "Auth write events" ON events FOR ALL USING (auth.uid() IS NOT NULL) WITH CHECK (auth.uid() IS NOT NULL); CREATE POLICY "Auth write posts" ON posts FOR ALL USING (auth.uid() IS NOT NULL) WITH CHECK (auth.uid() IS NOT NULL); CREATE POLICY "Auth write past_questions" ON past_questions FOR ALL USING (auth.uid() IS NOT NULL) WITH CHECK (auth.uid() IS NOT NULL); CREATE POLICY "Auth write gallery_photos" ON gallery_photos FOR ALL USING (auth.uid() IS NOT NULL) WITH CHECK (auth.uid() IS NOT NULL); INSERT INTO storage.buckets (id,name,public,file_size_limit,allowed_mime_types) VALUES ('gallery','gallery',true,10485760, ARRAY['image/jpeg','image/jpg','image/png','image/webp','image/gif']) ON CONFLICT (id) DO UPDATE SET public=true; INSERT INTO storage.buckets (id,name,public,file_size_limit,allowed_mime_types) VALUES ('past-questions','past-questions',true,20971520, ARRAY['application/pdf']) ON CONFLICT (id) DO UPDATE SET public=true; DROP POLICY IF EXISTS "Public read gallery" ON storage.objects; DROP POLICY IF EXISTS "Auth upload gallery" ON storage.objects; DROP POLICY IF EXISTS "Auth update gallery" ON storage.objects; DROP POLICY IF EXISTS "Auth delete gallery" ON storage.objects; DROP POLICY IF EXISTS "Public read pq" ON storage.objects; DROP POLICY IF EXISTS "Auth upload pq" ON storage.objects; DROP POLICY IF EXISTS "Auth update pq" ON storage.objects; DROP POLICY IF EXISTS "Auth delete pq" ON storage.objects; CREATE POLICY "Public read gallery" ON storage.objects FOR SELECT USING (bucket_id='gallery'); CREATE POLICY "Auth upload gallery" ON storage.objects FOR INSERT WITH CHECK (bucket_id='gallery' AND auth.uid() IS NOT NULL); CREATE POLICY "Auth update gallery" ON storage.objects FOR UPDATE USING (bucket_id='gallery' AND auth.uid() IS NOT NULL); CREATE POLICY "Auth delete gallery" ON storage.objects FOR DELETE USING (bucket_id='gallery' AND auth.uid() IS NOT NULL); CREATE POLICY "Public read pq" ON storage.objects FOR SELECT USING (bucket_id='past-questions'); CREATE POLICY "Auth upload pq" ON storage.objects FOR INSERT WITH CHECK (bucket_id='past-questions' AND auth.uid() IS NOT NULL); CREATE POLICY "Auth update pq" ON storage.objects FOR UPDATE USING (bucket_id='past-questions' AND auth.uid() IS NOT NULL); CREATE POLICY "Auth delete pq" ON storage.objects FOR DELETE USING (bucket_id='past-questions' AND auth.uid() IS NOT NULL);
🔐 Step 3 — Create Your Admin Account
Go to Supabase → Authentication → Settings → Email → turn OFF "Enable email confirmations".
Then go to Authentication → Users → Add user → Create new user. Enter your email and a strong password. Click Create User.
✅
Once created, use that email and password on the login screen. Your password is managed entirely by Supabase — it never appears in any code file.
